/* SlowQueryRadar */
SET LOCK_TIMEOUT 5000;

WITH RuntimeRows AS
(
	SELECT
		q.query_id AS QueryId,
		p.plan_id AS PlanId,
		CAST(rs.count_executions AS bigint) AS ExecutionCount,
		CAST(rs.avg_duration AS float) AS AvgDurationUs,
		CAST(rs.stdev_duration AS float) AS StdevDurationUs,
		CAST(rs.min_duration AS float) AS MinDurationUs,
		CAST(rs.max_duration AS float) AS MaxDurationUs,
		CAST(rs.avg_cpu_time AS float) AS AvgCpuUs,
		CAST(rs.avg_logical_io_reads AS float) AS AvgLogicalReads,
		CAST(rs.avg_physical_io_reads AS float) AS AvgPhysicalReads,
		CAST(rs.avg_logical_io_writes AS float) AS AvgLogicalWrites,
		CAST(rs.avg_rowcount AS float) AS AvgRowcount,
		CAST(rs.avg_query_max_used_memory AS float) AS AvgQueryMaxUsedMemory,
		rs.last_execution_time AS LastExecutionTime
	FROM sys.query_store_query AS q
	INNER JOIN sys.query_store_plan AS p ON p.query_id = q.query_id
	INNER JOIN sys.query_store_runtime_stats AS rs ON rs.plan_id = p.plan_id
	INNER JOIN sys.query_store_runtime_stats_interval AS rsi
		ON rsi.runtime_stats_interval_id = rs.runtime_stats_interval_id
	WHERE rsi.end_time >= @since
	  AND rsi.start_time <= @windowEnd
	  AND rs.count_executions > 0
),
Totals AS
(
	SELECT
		QueryId,
		SUM(ExecutionCount) AS TotalExecutions,
		SUM(AvgDurationUs * ExecutionCount) AS TotalDurationUs,
		SUM(AvgCpuUs * ExecutionCount) AS TotalCpuUs,
		SUM(AvgLogicalReads * ExecutionCount) AS TotalLogicalReads,
		SUM(AvgPhysicalReads * ExecutionCount) AS TotalPhysicalReads,
		SUM(AvgLogicalWrites * ExecutionCount) AS TotalLogicalWrites,
		SUM(AvgRowcount * ExecutionCount) AS TotalRows,
		SUM(AvgQueryMaxUsedMemory * ExecutionCount) AS TotalQueryMaxUsedMemory,
		-- Pooled variance uses E[x^2] - mean^2, where E[x^2] is built from
		-- each interval's count * (stdev^2 + average^2).
		SUM(ExecutionCount * (POWER(COALESCE(StdevDurationUs, 0), 2) + POWER(COALESCE(AvgDurationUs, 0), 2))) AS SumSqDuration,
		MIN(MinDurationUs) AS MinDurationUs,
		MAX(MaxDurationUs) AS MaxDurationUs,
		COUNT(DISTINCT PlanId) AS PlanCount,
		MAX(LastExecutionTime) AS LastExecutionTime
	FROM RuntimeRows
	GROUP BY QueryId
	HAVING SUM(ExecutionCount) >= @minExecutions
),
Metrics AS
(
	SELECT
		QueryId,
		TotalExecutions,
		TotalDurationUs,
		TotalDurationUs / NULLIF(TotalExecutions, 0) / 1000.0 AS AvgDurationMs,
		TotalDurationUs / 1000.0 AS TotalDurationMs,
		MinDurationUs / 1000.0 AS MinDurationMs,
		MaxDurationUs / 1000.0 AS MaxDurationMs,
		TotalCpuUs / NULLIF(TotalExecutions, 0) / 1000.0 AS AvgCpuMs,
		TotalCpuUs / 1000.0 AS TotalCpuMs,
		TotalLogicalReads / NULLIF(TotalExecutions, 0) AS AvgLogicalReads,
		TotalLogicalReads,
		TotalPhysicalReads,
		TotalLogicalWrites,
		TotalRows,
		TotalQueryMaxUsedMemory,
		PlanCount,
		LastExecutionTime,
		CASE WHEN TotalExecutions > 1
			THEN SumSqDuration / TotalExecutions - POWER(TotalDurationUs / TotalExecutions, 2)
			ELSE 0 END AS VarianceUs2
	FROM Totals
),
FinalMetrics AS
(
	SELECT *, SQRT(CASE WHEN VarianceUs2 < 0 THEN 0 ELSE VarianceUs2 END) / 1000.0 AS StandardDeviationMs
	FROM Metrics
)
SELECT TOP (@topN)
	m.QueryId,
	CONVERT(varchar(18), q.query_hash, 1) AS QueryHash,
	q.object_id AS ObjectId,
	CASE WHEN q.object_id IS NULL THEN NULL ELSE OBJECT_SCHEMA_NAME(q.object_id) + N'.' + OBJECT_NAME(q.object_id) END AS ObjectName,
	qt.query_sql_text AS SqlText,
	m.PlanCount,
	m.TotalExecutions,
	m.AvgDurationMs,
	m.TotalDurationMs,
	m.MinDurationMs,
	m.MaxDurationMs,
	m.StandardDeviationMs,
	m.TotalCpuMs,
	m.AvgCpuMs,
	m.TotalLogicalReads,
	m.AvgLogicalReads,
	m.TotalPhysicalReads,
	m.TotalLogicalWrites,
	m.TotalRows,
	m.TotalQueryMaxUsedMemory,
	m.LastExecutionTime,
	p.last_compile_start_time AS LastCompileStartTime
FROM FinalMetrics AS m
INNER JOIN sys.query_store_query AS q ON q.query_id = m.QueryId
INNER JOIN sys.query_store_query_text AS qt ON qt.query_text_id = q.query_text_id
OUTER APPLY (SELECT MAX(last_compile_start_time) AS last_compile_start_time FROM sys.query_store_plan WHERE query_id = m.QueryId) AS p
WHERE (@excludeNoise = 0 OR (
	q.is_internal_query = 0
	AND qt.query_sql_text NOT LIKE '%sys.query_store_%'
	AND qt.query_sql_text NOT LIKE '%sys.dm_%'
	AND qt.query_sql_text NOT LIKE '%UPDATE STATISTICS%'
	AND qt.query_sql_text NOT LIKE '%ALTER INDEX%'
	AND qt.query_sql_text NOT LIKE '%DBCC %'
	AND qt.query_sql_text NOT LIKE '%BACKUP %'
	AND qt.query_sql_text NOT LIKE '%sp_whoisactive%'
	AND LTRIM(qt.query_sql_text) NOT LIKE '/* SlowQueryRadar */%'))
  AND (@minAvgDurationMs IS NULL OR m.AvgDurationMs >= @minAvgDurationMs)
/*ORDER_BY*/;

/// <summary>Collects read-only metrics from SQL Server Query Store.</summary>
public sealed class QueryStoreCollector : IQueryStoreCollector
{
    private const string StateSql = "SELECT actual_state_desc AS ActualStateDesc, query_capture_mode_desc AS QueryCaptureModeDesc, wait_stats_capture_mode_desc AS WaitStatsCaptureModeDesc FROM sys.database_query_store_options;";
    private const string LegacySql = "SELECT TOP (@topN) q.query_id AS QueryId, qt.query_sql_text AS SqlText, SUM(rs.count_executions) AS ExecCount, AVG(rs.avg_duration) / 1000.0 AS AvgDurationMs, SUM(rs.avg_cpu_time * rs.count_executions) / 1000.0 AS TotalCpuMs, SUM(rs.avg_logical_io_reads * rs.count_executions) AS TotalLogicalReads, SUM(rs.avg_cpu_time * rs.count_executions) AS ImpactScore, TRY_CAST(p.query_plan AS NVARCHAR(MAX)) AS ExecutionPlanXml, MAX(rsi.end_time) AS LastExecutionUtc FROM sys.query_store_query AS q INNER JOIN sys.query_store_query_text AS qt ON q.query_text_id = qt.query_text_id INNER JOIN sys.query_store_plan AS p ON q.query_id = p.query_id INNER JOIN sys.query_store_runtime_stats AS rs ON p.plan_id = rs.plan_id INNER JOIN sys.query_store_runtime_stats_interval AS rsi ON rs.runtime_stats_interval_id = rsi.runtime_stats_interval_id WHERE rsi.start_time > @since GROUP BY q.query_id, qt.query_sql_text, p.query_plan ORDER BY ImpactScore DESC;";
    private const string PlanXmlSql = "SELECT TRY_CONVERT(xml, p.query_plan) FROM sys.query_store_plan AS p WHERE p.query_id = @queryId AND p.plan_id = @planId;";

    private readonly QueryStoreOptions _options;
    private readonly Func<DbConnection> _connectionFactory;
    private readonly ILogger<QueryStoreCollector> _logger;

    /// <summary>Initializes a collector with configured SQL connections.</summary>
    public QueryStoreCollector(IOptions<QueryStoreOptions> options) : this(options, null, NullLogger<QueryStoreCollector>.Instance) { }
    /// <summary>Initializes a collector with a connection factory.</summary>
    public QueryStoreCollector(IOptions<QueryStoreOptions> options, Func<DbConnection>? connectionFactory) : this(options, connectionFactory, NullLogger<QueryStoreCollector>.Instance) { }
    /// <summary>Initializes a collector with a connection factory and logger.</summary>
    public QueryStoreCollector(IOptions<QueryStoreOptions> options, Func<DbConnection>? connectionFactory, ILogger<QueryStoreCollector> logger)
    {
        ArgumentNullException.ThrowIfNull(options);
        ArgumentNullException.ThrowIfNull(logger);
        _options = options.Value ?? throw new ArgumentException("Query Store options are required.", nameof(options));
        _connectionFactory = connectionFactory ?? (() => new SqlConnection(_options.ConnectionString));
        _logger = logger;
    }

    /// <inheritdoc />
    public async Task<CollectResult> CollectAsync(CollectOptions options, CancellationToken ct)
    {
        ArgumentNullException.ThrowIfNull(options);
        options.Validate();
        var end = DateTimeOffset.UtcNow;
        var start = end.Subtract(options.Lookback);
        await using var connection = _connectionFactory();
        await connection.OpenAsync(ct).ConfigureAwait(false);
        var commandTimeout = _options.CommandTimeoutSeconds is > 0 and <= 30 ? _options.CommandTimeoutSeconds : 30;
        var state = await connection.QueryFirstOrDefaultAsync<QueryStoreStateDto>(new CommandDefinition(StateSql, commandTimeout: commandTimeout, cancellationToken: ct)).ConfigureAwait(false)
            ?? throw new QueryStoreDisabledException("Query Store state could not be read. Ask a DBA to verify Query Store permissions.");
        if (string.Equals(state.ActualStateDesc, "OFF", StringComparison.OrdinalIgnoreCase))
            throw new QueryStoreDisabledException($"Query Store is not enabled. Ask a DBA to run: ALTER DATABASE {connection.Database} SET QUERY_STORE = ON;");

        var warnings = new List<string>();
        if (string.Equals(state.ActualStateDesc, "READ_ONLY", StringComparison.OrdinalIgnoreCase)) warnings.Add("Query Store is READ_ONLY; new runtime data may not be captured.");
        if (string.Equals(state.QueryCaptureModeDesc, "NONE", StringComparison.OrdinalIgnoreCase)) warnings.Add("Query Store query capture mode is NONE.");
        var sql = LoadResource("TopSlowQueries.sql");
        var rows = (await connection.QueryAsync<CollectionDto>(new CommandDefinition(sql, new
        {
            since = start.UtcDateTime, windowEnd = end.UtcDateTime, top = options.Top,
            minExecutions = options.MinExecutions, minAvgDurationMs = options.MinAvgDurationMs,
            excludeNoise = options.ExcludeNoise ? 1 : 0, rankBy = (int)options.RankBy
        }, commandTimeout: commandTimeout, cancellationToken: ct)).ConfigureAwait(false)).AsList();
        var plans = await LoadPlansAsync(connection, rows.Select(x => x.QueryId), start, end, ct).ConfigureAwait(false);
        var captured = DateTimeOffset.UtcNow;
        var snapshots = rows.Select(row => Map(row, plans.Where(plan => plan.QueryId == row.QueryId).ToArray(), captured, options.IncludePlanXml)).ToArray();
        _logger.LogInformation("Collected {Count} weighted queries with rank {RankBy} and lookback {Lookback}", snapshots.Length, options.RankBy, options.Lookback);
        return new CollectResult(snapshots, warnings, start, end, state.ActualStateDesc, captured);
    }

    /// <inheritdoc />
    public async Task<IReadOnlyList<SlowQuerySnapshot>> CollectTopSlowQueriesAsync(int topN, TimeSpan lookback, CancellationToken ct)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(topN);
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(lookback.TotalSeconds);
        await using var connection = _connectionFactory();
        await connection.OpenAsync(ct).ConfigureAwait(false);
        var state = await connection.QuerySingleAsync<string>(new CommandDefinition("SELECT actual_state_desc FROM sys.database_query_store_options;", commandTimeout: 30, cancellationToken: ct)).ConfigureAwait(false);
        if (!string.Equals(state, "READ_WRITE", StringComparison.OrdinalIgnoreCase) && !string.Equals(state, "READ_ONLY", StringComparison.OrdinalIgnoreCase))
            throw new QueryStoreDisabledException($"Query Store is not enabled. Ask a DBA to run: ALTER DATABASE {connection.Database} SET QUERY_STORE = ON;");
        var rows = await connection.QueryAsync<LegacyCollectionDto>(new CommandDefinition(LegacySql, new { since = DateTime.UtcNow.Subtract(lookback), topN }, commandTimeout: 30, cancellationToken: ct)).ConfigureAwait(false);
        var captured = DateTimeOffset.UtcNow;
        return rows.Select(row => new SlowQuerySnapshot { QueryId = row.QueryId, SqlText = row.SqlText, NormalizedSqlHash = ComputeNormalizedSqlHash(row.SqlText), TotalExecutions = row.ExecCount, AvgDurationMs = row.AvgDurationMs, TotalCpuMs = row.TotalCpuMs, TotalLogicalReads = row.TotalLogicalReads, ExecutionPlanXml = row.ExecutionPlanXml, LastExecutionTime = row.LastExecutionUtc, CapturedAtUtc = captured }).ToArray();
    }

    /// <inheritdoc />
    public async Task<string?> GetPlanXmlAsync(long queryId, long planId, CancellationToken ct)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(queryId);
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(planId);
        await using var connection = _connectionFactory();
        await connection.OpenAsync(ct).ConfigureAwait(false);
        return await connection.ExecuteScalarAsync<string?>(new CommandDefinition(PlanXmlSql, new { queryId, planId }, commandTimeout: 30, cancellationToken: ct)).ConfigureAwait(false);
    }

    /// <inheritdoc />
    public async Task<SlowQuerySnapshot?> GetByQueryIdAsync(long queryId, CancellationToken ct)
    {
        var result = await CollectAsync(new CollectOptions { Top = 500, MinExecutions = 1 }, ct).ConfigureAwait(false);
        return result.Queries.FirstOrDefault(query => query.QueryId == queryId);
    }

    private async Task<IReadOnlyList<PlanDto>> LoadPlansAsync(DbConnection connection, IEnumerable<long> ids, DateTimeOffset start, DateTimeOffset end, CancellationToken ct)
    {
        var queryIds = ids.Distinct().ToArray();
        if (queryIds.Length == 0) return Array.Empty<PlanDto>();
        var rows = await connection.QueryAsync<PlanDto>(new CommandDefinition(LoadResource("PlanEvidence.sql"), new { queryIds, since = start.UtcDateTime, windowEnd = end.UtcDateTime }, commandTimeout: 30, cancellationToken: ct)).ConfigureAwait(false);
        return rows.AsList();
    }

    private static SlowQuerySnapshot Map(CollectionDto row, IReadOnlyList<PlanDto> plans, DateTimeOffset captured, bool includeXml) => new()
    {
        QueryId = row.QueryId, QueryHash = row.QueryHash, ObjectId = row.ObjectId, ObjectName = null, SqlText = row.SqlText,
        NormalizedSqlHash = ComputeNormalizedSqlHash(row.SqlText), TotalExecutions = row.TotalExecutions, AvgDurationMs = row.AvgDurationMs,
        TotalDurationMs = row.TotalDurationMs, MinDurationMs = row.MinDurationMs, MaxDurationMs = row.MaxDurationMs, StandardDeviationMs = row.StandardDeviationMs,
        TotalCpuMs = row.TotalCpuMs, AvgCpuMs = row.AvgCpuMs, TotalLogicalReads = row.TotalLogicalReads, AvgLogicalReads = row.AvgLogicalReads,
        TotalPhysicalReads = row.TotalPhysicalReads, TotalLogicalWrites = row.TotalLogicalWrites, TotalRows = row.TotalRows, PlanCount = row.PlanCount,
        LastExecutionTime = row.LastExecutionTime, LastCompileStartTime = row.LastCompileStartTime,
        Plans = plans.Select(plan => new PlanSummary(plan.PlanId, plan.IsForcedPlan, plan.LastExecutionTime, plan.ExecutionCount, plan.AvgDurationMs, plan.MaxDurationMs, plan.StandardDeviationMs, null)).ToArray(),
        CapturedAtUtc = captured
    };

    private static string LoadResource(string name)
    {
        using var stream = Assembly.GetExecutingAssembly().GetManifestResourceStream($"SlowQueryRadar.Collector.Sql.{name}")
            ?? throw new InvalidOperationException($"Embedded SQL resource '{name}' was not found.");
        using var reader = new StreamReader(stream);
        return reader.ReadToEnd();
    }

    private static string ComputeNormalizedSqlHash(string sqlText)
    {
        var withoutComments = Regex.Replace(sqlText, @"(--[^\r\n]*|/\*.*?\*/)", " ", RegexOptions.Singleline);
        var withoutLiterals = Regex.Replace(withoutComments, @"'(?:''|[^'])*'", " ");
        return Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(Regex.Replace(withoutLiterals.ToLowerInvariant(), @"\s+", " ").Trim())));
    }
}
